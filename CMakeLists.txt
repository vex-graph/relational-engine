cmake_minimum_required(VERSION 3.20)

# CLion code-model adapter only. b remains the actual build entry point.
if(APPLE)
    set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Supported macOS architecture")
    set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0 CACHE STRING "Supported macOS floor")
endif()
project(relational_engine_ide LANGUAGES C)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

set(VEXSPOKE_SOURCE_DIR "" CACHE PATH "Optional local Vexspoke src for C reference headers")
foreach(part io relational reflection)
    file(GLOB part_sources CONFIGURE_DEPENDS "src/${part}/*.c" "src/${part}/*.h")
    add_library(reference_${part} OBJECT EXCLUDE_FROM_ALL ${part_sources})
    set_target_properties(reference_${part} PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED YES C_EXTENSIONS YES)
    target_include_directories(reference_${part} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
    if(VEXSPOKE_SOURCE_DIR)
        target_include_directories(reference_${part} PRIVATE "${VEXSPOKE_SOURCE_DIR}")
    endif()
    if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(reference_${part} PRIVATE -Wall -Wextra -Werror)
        if(APPLE)
            target_compile_options(reference_${part} PRIVATE -mcpu=apple-m1)
        endif()
    endif()
endforeach()

message(STATUS "IDE metadata only: reference targets are excluded from the default build; use b for actual builds")
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/nio/mem.h")
    message(STATUS "C reference gap: src/nio/mem.h is missing; unresolved dependencies are not stubbed")
endif()
