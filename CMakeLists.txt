cmake_minimum_required(VERSION 3.20)

# CLion code-model adapter only. b remains the actual build entry point.
if(APPLE)
    set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Supported macOS architecture")
    set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0 CACHE STRING "Supported macOS floor")
endif()
project(relational_engine_ide LANGUAGES C)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

set(VEXSPOKE_SOURCE_DIR "" CACHE PATH "Optional local Vexspoke src for CPU leaf headers")
# R2 production targets: canonical nio/io storage owners plus the engine-owned
# native span search (name_search, run through Cargo). These are production;
# relational/reflection stay reference-only so they never shadow Vexspoke's
# canonical headers.
foreach(part nio io search)
    file(GLOB_RECURSE part_sources CONFIGURE_DEPENDS "src/${part}/*.c" "src/${part}/*.h")
    if(APPLE AND part STREQUAL "io")
        enable_language(OBJC)
        file(GLOB_RECURSE part_objc CONFIGURE_DEPENDS "src/${part}/*.m")
        list(APPEND part_sources ${part_objc})
    endif()
    add_library(engine_${part} OBJECT EXCLUDE_FROM_ALL ${part_sources})
    set_target_properties(engine_${part} PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED YES C_EXTENSIONS YES)
    target_include_directories(engine_${part} PRIVATE "src")
    if(VEXSPOKE_SOURCE_DIR)
        target_include_directories(engine_${part} PRIVATE "${VEXSPOKE_SOURCE_DIR}")
    endif()
    if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(engine_${part} PRIVATE -Wall -Wextra -Werror)
        if(APPLE)
            target_compile_options(engine_${part} PRIVATE -mcpu=apple-m1)
        endif()
    endif()
endforeach()

foreach(part relational reflection)
    file(GLOB_RECURSE part_sources CONFIGURE_DEPENDS "src/${part}/*.c" "src/${part}/*.h")
    add_library(reference_${part} OBJECT EXCLUDE_FROM_ALL ${part_sources})
    set_target_properties(reference_${part} PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED YES C_EXTENSIONS YES)
    target_include_directories(reference_${part} PRIVATE "src")
    if(VEXSPOKE_SOURCE_DIR)
        target_include_directories(reference_${part} PRIVATE "${VEXSPOKE_SOURCE_DIR}")
    endif()
    if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(reference_${part} PRIVATE -Wall -Wextra -Werror)
        if(APPLE)
            target_compile_options(reference_${part} PRIVATE -mcpu=apple-m1)
        endif()
    endif()
endforeach()

message(STATUS "IDE metadata only: reference targets are excluded from the default build; use b for actual builds")
if(NOT EXISTS "src/nio/mem.h")
    message(STATUS "C reference gap: src/nio/mem.h is missing; unresolved dependencies are not stubbed")
endif()
